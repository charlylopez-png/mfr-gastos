⚠ Configura tu Supabase antes de usar la app. Edita las variables SUPABASE_URL y SUPABASE_ANON_KEY al inicio del script, y crea las tablas con el SQL de abajo.
-- Ejecuta este SQL en tu proyecto Supabase (SQL Editor) create table gastos ( id uuid primary key default gen_random_uuid(), tipo text not null check (tipo in ('mantenimiento','alquiler','compra')), fecha_envio date, fecha_recepcion date, año integer, mes text, code text, elemento text, descripcion text, averia text, produccion text, proveedor text, coste numeric(12,2), refact numeric(12,2), albaran_num text, factura_num text, num_pedido text, dingo boolean default false, notas text, created_at timestamptz default now(), created_by uuid references auth.users(id) ); create table documentos ( id uuid primary key default gen_random_uuid(), gasto_id uuid references gastos(id) on delete cascade, tipo text not null check (tipo in ('presupuesto','albaran','factura')), nombre_archivo text, storage_path text, url text, created_at timestamptz default now() ); -- Habilitar RLS alter table gastos enable row level security; alter table documentos enable row level security; -- Políticas: usuarios autenticados pueden hacer todo create policy "auth_all_gastos" on gastos for all to authenticated using (true) with check (true); create policy "auth_all_documentos" on documentos for all to authenticated using (true) with check (true); -- Bucket para documentos (crear en Storage > New bucket, nombre: "documentos", public: false) -- Política de storage: create policy "auth_storage" on storage.objects for all to authenticated using (bucket_id = 'documentos') with check (bucket_id = 'documentos');
